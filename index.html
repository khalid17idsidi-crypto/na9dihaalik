-- 1. جدول المستخدمين create table public.users ( id uuid default gen_random_uuid() primary key, phone text unique not null, full_name text not null, role text check (role in ('client', 'courier')) not null, created_at timestamp with time zone default timezone('utc'::text, now()) ); -- 2. جدول بيانات الموصلين create table public.courier_profiles ( id uuid default gen_random_uuid() primary key, user_id uuid references public.users(id) on delete cascade, vehicle_type text default 'moto', profile_photo text not null, -- الصورة الشمسية إلزامية status text default 'pending' check (status in ('pending', 'approved')) ); -- 3. جدول الطلبات create table public.delivery_orders ( id uuid default gen_random_uuid() primary key, client_id uuid references public.users(id) on delete cascade, pickup_address text not null, delivery_address text not null, package_photo text not null, -- صورة الطرد إلزامية offered_price numeric default 25, legal_terms_accepted boolean not null check (legal_terms_accepted = true), status text default 'pending' check (status in ('pending', 'accepted', 'delivered')), created_at timestamp with time zone default timezone('utc'::text, now()) );
